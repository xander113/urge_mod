URGE_grant_military_supremacy_effect = {
	if = {
		limit = { NOT = { has_idea = URGE_military_supremacy } }
		add_ideas = URGE_military_supremacy
	}
	every_army_leader = {
		if = { limit = { NOT = { has_trait = URGE_supreme_commander } }
			add_unit_leader_trait = URGE_supreme_commander }
	}
	every_navy_leader = {
		if = { limit = { NOT = { has_trait = URGE_grand_admiral } }
			add_unit_leader_trait = URGE_grand_admiral }
	}
	army_experience = 9999
	navy_experience = 9999
	air_experience = 9999
	add_political_power = 9999
	add_equipment_to_stockpile = { type = convoy_1 amount = 9999 producer = ROOT }
}

URGE_remove_military_supremacy_effect = {
	if = {
		limit = { has_idea = URGE_military_supremacy }
		remove_ideas = URGE_military_supremacy
	}
}
